#include<iostream>
#include<vector>
#include<algorithm>
using namespace std;

int main(){
	int n;
	cin>>n;
	vector<int>a;
	for (int i = 0 ; i < n ; ++i){
		int x ;
		cin >> x;
		a.push_back(x);
		}
	sort (a.begin() , a.end() );
	for (int y = 0 ; y < n ; ++y ){
		cout << a[y] << " " ;
		}
	return 0;
	}
