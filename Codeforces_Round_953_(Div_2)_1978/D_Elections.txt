ll sum(ll n) {
    return (n * (n + 1)) / 2;
}

void accepted() {
    ll n, c;
    cin >> n >> c;
    vector<ll> v, prfx(n);
    ll mx = -1 , indx;
    for (int i = 0; i < n; i++) {
        ll inp;
        cin >> inp;
        if (i == 0)
            inp += c;
        v.pb(inp);
        prfx[i] += inp;
        if (i != 0)
            prfx[i] += prfx[i - 1];
        if ( inp > mx ){
            mx = inp ;
            indx = i ;
        }
    }
    if (v[0] == mx) cout << 0;
    else cout << 1;
    cout << " ";
    for (int i = 1; i < n; i++) {
        if ( i == indx ){
            cout << 0 << " " ;
            continue;
            
        }
        ll x = v[i];
        x += prfx[i - 1];
        if (x >= mx) cout << i;
        else cout << i + 1;
        cout << " ";
    }
}

int main() {
    ASSAF;
    int exe = 1;
    cin >> exe;
    while (exe--) {
        accepted();
        cout << "\n";
    }
    return 0;
}
