ll sum(ll n ){
    return ( n*(n+1) )/2;
}

void accepted(){
    ll n , a  ,b ;
    cin >> n >> a >> b ;
    ll k = max((b-a),0ll);
    if ( k > n )
        k = n ;
    ll profit = (n-k)*a;
    if ( k > 0 )
        profit += (sum(b) - sum(b-k));
    cout << profit ;
 }
