#include<bits/stdc++.h>
using namespace std;
int n, m;
bool valid(int x , int y ) {
	if (x > -1 && x < n && y > -1 && y < m)
		return 1;
	return 0;
}
const int N = 101, M = 101;
int main() {
	cin >> n >> m;
	char arr[N][M];
	for (int i = 0; i < n; i++) {
		for (int j = 0; j < m; j++) {
			cin >> arr[i][j];
		}
	}
	string s = "BW";
	int x = 0;
	for (int i = 0; i < n; i++) {
		x = (i & 1 ? 1 : 0);
		for (int j = 0 ; j < m; j++ , x^=1 ) {
			if (arr[i][j] == '-') continue;
			arr[i][j] = s[x];
		}
	}
	for (int i = 0; i < n; i++) {
		for (int j = 0; j < m; j++) {
			cout << arr[i][j];
		}
		cout << '\n';
	}
	return 0;
}
