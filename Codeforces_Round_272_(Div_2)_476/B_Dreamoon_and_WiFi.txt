#include <bits/stdc++.h>

#define ASSAF ios::sync_with_stdio(0), cin.tie(0), cout.tie(0)
using namespace std;

// assaf
//  /﹋\
// (҂`_´)
// <,︻╦╤─ - - - -  accepted
// /﹋\

string s1, s2;
double tries = 0 , valid = 0;
int res1 = 0;
void solve(int idx = 0 , int res2 = 0 )
{
    if ( idx == s2.size() ){
        tries++;
        if ( res1 == res2 ){
            valid++;
        }
        return;
    }

    if ( s2[idx] == '+' ) {
        solve(idx+1,res2+1);
    }else if ( s2[idx] == '-' ) {
        solve(idx+1,res2-1);
    }else{
        solve(idx+1,res2+1);
        solve(idx+1,res2-1);
    }
}

int main()
{
    int t = 1;
    // cin >> t;
    while (t--)
    {
        cin >> s1 >> s2;
        for (int i = 0; i < s1.size(); i++)
        {
            if ( s1[i] == '+' ) res1++;
            else res1--;
        }
        solve();
        double ans = valid/tries;
        cout << fixed << setprecision(12) << ans ;
        // cout << "\n";
    }
    return 0;
}
