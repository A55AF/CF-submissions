#include<bits/stdc++.h>
using namespace std;
int main(){
	int n ;
	cin >> n ;
	long long arr[n] ;
	for ( int i = 0 ; i < n ; i++ )
		cin >> arr[i] ;
	sort ( arr , arr + n );
	long long result = arr[0] + arr[n-1];
	int cnt = 0;
	for ( int i = 0 ; i < n ; i++ ){
		for ( int j = i+1 ; j < n ; j++ ){
			if ( arr[i] + arr[j] == result )
				cnt++;
		}
	}
	cout << cnt ;
	return 0;
}
