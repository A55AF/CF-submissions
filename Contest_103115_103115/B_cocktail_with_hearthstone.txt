const ll mod = 1e9 + 7;
const int N = 5e5 + 7;
ll fact[N];
ll add(ll x, ll y) { return ((x % mod) + (y % mod)) % mod; }
ll sub(ll x, ll y) { return ((x % mod) - (y % mod) + mod) % mod; }
ll mult(ll x, ll y) { return ((x % mod) * (y % mod)) % mod; }
ll fpow(ll b, ll pw)
{
    ll ans = 1;
    while (pw > 0)
    {
        if (pw % 2 == 1)
            ans = 1LL * mult(ans, b);
        b = mult(b, b);
        pw = pw / 2;
    }
    return ans;
}
ll inv(ll x) { return fpow(x, mod - 2); }
void factorial()
{
    fact[0] = fact[1] = 1;
    for (int i = 2; i < N; i++)
        fact[i] = mult(fact[i - 1], i);
}
ll divis(ll x, ll y) { return mult(x, inv(y)); }
ll nCr(ll n, ll r)
{
    if (n < r)
        return 0;
    return divis(fact[n], mult(fact[r], fact[n - r]));
}
ll nPr(ll n, ll r)
{
    if (n == 0 || n == r || n < r)
        return 0;
    return divis(fact[n], fact[n - r]);
}
ll snb(ll n, ll k) { return nCr(n + k - 1, k - 1); }

void accepted()
{
    ll n, m, q;
    cin >> n >> m >> q;
    while (q--)
    {
        ll x, y;
        cin >> x >> y;
        if (x == n)
            cout << mult(fpow(2, n + m - x - y), snb(y, x));
        else if (y == m)
            cout << mult(fpow(2, n + m - x - y), snb(x, y));
        cout << '\n';
    }
}

int main()
{
    ASSAF;
    factorial();
    int exe = 1;
    // cin >> exe;
    while (exe--)
    {
        accepted();
        // cout << "\n";
    }
    return 0;
}
