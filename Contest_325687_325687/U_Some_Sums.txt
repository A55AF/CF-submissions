#include<bits/stdc++.h>
using namespace std;
int main(){
    long long n ,a , b;
    cin >> n >> a >> b ;
    long long sum = 0;
    for ( int i = 1 ; i <= n ; ++i ){ 
        int x = 0;
        int y = i;
        while ( y > 0 ){
            x += y%10;
            y /= 10;
        }
        if ( x >= a && x <= b ){
            sum += i;
        }
    }
    cout << sum;
    return 0;
}
