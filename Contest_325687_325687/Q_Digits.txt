#include <iostream>
#include <vector>
#include <algorithm>
 
using namespace std;
 
int main() {
    int t;
    cin >> t ;
    int a[t];
    for ( int i = 0 ; i < t ; ++i ){
        cin >> a[i] ;
    }
    for ( int j = 0 ; j < t ; ++j ){
        while ( a[j] >= 0 ){
            int x = a[j] % 10;
            cout << x << " ";
            a[j] /= 10;
            if (a[j]==0)
                --a[j];
 
        }
        if ( j == t - 1 ){
            return 0;
        }
        cout << endl;
    }
 
    return 0;
}
