#include<bits/stdc++.h>
using namespace std;

int main(){
    int n;
    cin >> n;
    int m , cnt;
    if ( n == 2 )
        cout << "2" ;
    if ( n >= 3 ){
        cout << "2 3 ";
    }
    for ( int i = 4; i <= n; ++i){
        m = i / 2;
        cnt = 0;
        for ( int y = 2 ; y <= m ; ++y){
            if ( i % y == 0 ){
                cnt++;
            }
            else{
                continue;
            }
        }
        if ( cnt == 0 ){
            cout << i << " ";
        }
    }
    return 0;
}
