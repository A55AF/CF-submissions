#include<bits/stdc++.h>
using namespace std;

int x , y;

void switch_num(){
    int c = x;
    x = y;
    y = c;
}

int sum( int x, int y){
    int res = 0;
    for ( int i = (x + 1) ; i < y ; ++i ){
        if ( i % 2 != 0 ){
            res += i;
        }
    }
    return res;
}

int main() {
    int t;
    cin >> t ;
    while ( t-- ){
        cin >> x >> y;
        if ( y < x ){
            switch_num();
        }
        cout << sum(x, y) << endl ;
    }
	return 0;
}
