#include<bits/stdc++.h>
#define ASSAF ios::sync_with_stdio(0),cin.tie(0),cout.tie(0)
using namespace std;

// assaf
//  /﹋\
// (҂`_´)
// <,︻╦╤─ - - - -  accepted
// /﹋\
↓↓↓ START FROM HERE ↓↓↓

int main(){
    ASSAF;
    int t ;
    cin >> t ;
    while ( t-- ){
        long long n ; cin >> n ;
        long long arr[n] ;
        for ( int i = 0 ; i < n ;i++ )
            cin >> arr[i] ;
        map<pair<long long,long long> , long long > ab , ac , bc ;
        map<vector<long long>,long long> all ;
        long long res = 0;
        for ( int i = 0 ; i < n -2 ; i++ ){
            long long a = arr[i] , b = arr[i+1] , c = arr[i+2] ;
            res += ac[{a,c}] - all[{a, b,c}];
            res += ab[{a,b}] - all[{a,b,c}];
            res += bc[{b,c}] - all[{a,b,c}];
            ac[{a,c}]++,ab[{a,b}]++,bc[{b,c}]++,all[{a,b,c}]++;
        }
        cout << res << '\n' ;
    }
    return 0;
}
