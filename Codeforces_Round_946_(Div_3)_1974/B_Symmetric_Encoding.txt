#include <bits/stdc++.h>

#define ASSAF ios::sync_with_stdio(0), cin.tie(0), cout.tie(0)
using namespace std;

// assaf
//  /﹋\
// (҂`_´)
// <,︻╦╤─ - - - -  accepted
// /﹋\

int main()
{
    ASSAF;
    int t ;
    cin >> t ;
    while ( t-- ) {
        int n ;
        cin >> n ;
        string s ;
        cin >> s ;
        string r;
        map<char,char> f;
        bool vis[30] = {};
        for ( int i = 0 ; i < n ; i++ ) {
            if ( !vis[s[i]-'a']) {
                r.push_back(s[i]);
                vis[s[i]-'a'] = 1;
            }
        }
        sort(r.begin(),r.end());
        for ( int i = 0 , j = r.size()-1; i < r.size() ; i++ , j-- ) {
            f[r[i]] = r[j];
        }
        
        for ( int i = 0 ; i < n ; i++)
            cout << f[s[i]] ;
        cout << '\n' ;
    }
    return 0;
}
