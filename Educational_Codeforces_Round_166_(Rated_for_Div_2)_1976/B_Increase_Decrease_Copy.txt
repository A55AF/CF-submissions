void accepted() {
    ll n ;
    cin >> n ;
    ll a[n] , b[n+1] ;
    for ( int i = 0 ; i < n ; i++ ){
        cin >> a[i] ;
    }
    for ( int j = 0 ; j < n+1 ; j++ )
        cin >> b[j] ;
    bool flag = 0 ;
    ll mnD = LONG_LONG_MAX;
    ll ans = 0 ;
    for ( int i = 0 ; i < n ; i++ ){
        ans+=abs(a[i]-b[i]);
        if ( b[n] >= min(a[i],b[i]) && b[n] <= max(a[i],b[i]))
            flag = 1;
        else{
            mnD = min({mnD,abs(b[n]-a[i]) , abs(b[n]-b[i])});
        }
    }
    if ( flag )
        ans++;
    else{
        ans += mnD + 1 ;
    }
    cout << ans ;
}

int main() {
    ASSAF;
    int exe = 1;
    cin >> exe;
    while (exe--) {
        accepted();
        cout << "\n";
    }
    return 0;
}
