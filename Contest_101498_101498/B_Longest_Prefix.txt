#include<bits/stdc++.h>
using namespace std;

int main()
{
    int t;
    cin >> t;
    while (t--) {
        string a, b;
        cin >> a >> b;
        map<char, int> first_str, second_str;
        for (int i = 0; i < a.size(); i++) {
            first_str[a[i]]++;
        }
        for (int j = 0; j < b.size(); j++) {
            second_str[b[j]]++;
        }

        int answer = 0;
        for (int i = 0; i < min(a.size(),b.size()); i++) {
            if (second_str[a[i]] > 0) {
                answer++;
                first_str[a[i]]--;
                second_str[a[i]]--;
            }else
                break;
        }
        cout << answer << '\n';
    }
    return 0;
}
